if(NOT TPL_ENABLE_MPI OR MPI_EXEC_MAX_NUMPROCS LESS 4)
    return()
endif()
foreach(model IN ITEMS navierStokes fsi newmark artery)
    set(variable_test_name ${PACKAGE_NAME}_${model}_variableTimeStep_restart)
    if(model STREQUAL "navierStokes")
        set(input_case navierStokes_2D_3D_bfs)
        set(driver navierStokes_2D_3D_bfs)
        set(meshes BFS2d_1600.mesh)
    elseif(model STREQUAL "fsi")
        set(input_case fsi_2D_turek)
        set(driver fsi_2D_turek)
        set(meshes dfg_fsi_fluid_h004.mesh dfg_fsi_solid_h004.mesh)
    elseif(model STREQUAL "artery")
        set(input_case fsi_3D_arterialSegment)
        set(driver fsi_3D_arterialSegment)
        set(meshes fsi_fluid_length_0_5_mm.mesh fsi_solid_length_0_5_mm.mesh)
    else()
        set(input_case unsteadyNonLinElasticity_restart)
        set(driver unsteadyNonLinElasticity_restart)
        set(meshes square_h02.mesh)
    endif()
    configure_file(runVariableRestart.cmake.in ${variable_test_name}.cmake @ONLY)
    add_test(NAME ${variable_test_name} COMMAND ${CMAKE_COMMAND}
        "-DEXECUTABLE=$<TARGET_FILE:${PACKAGE_NAME}_${driver}>"
        -P ${CMAKE_CURRENT_BINARY_DIR}/${variable_test_name}.cmake)
    set_tests_properties(${variable_test_name} PROPERTIES PROCESSORS 4 LABELS ${PACKAGE_NAME}
        REQUIRED_FILES "$<TARGET_FILE:${PACKAGE_NAME}_${driver}>")
endforeach()
